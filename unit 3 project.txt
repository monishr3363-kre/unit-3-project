#include <stdio.h>
#include <string.h>

#define MAX 100

struct Student
{
    int rollNo;
    char name[50];
    float cutoffMark;
    char city[50];
    char dream[100];
};

int main()
{
    struct Student s[MAX];
    int n, i;

    printf("====================================\n");
    printf("   STUDENT DETAILS MANAGEMENT SYSTEM\n");
    printf("====================================\n");

    printf("Enter number of students: ");
    if (scanf("%d", &n) != 1 || n <= 0 || n > MAX)
    {
        printf("Invalid number of students.\n");
        return 1;
    }

    for (i = 0; i < n; i++)
    {
        printf("\nEnter details of Student %d\n", i + 1);

        printf("Enter Roll Number: ");
        scanf("%d", &s[i].rollNo);
        getchar();

        printf("Enter Name: ");
        fgets(s[i].name, sizeof(s[i].name), stdin);
        s[i].name[strcspn(s[i].name, "\n")] = '\0';

        printf("Enter Cut-off Mark: ");
        scanf("%f", &s[i].cutoffMark);
        getchar();

        printf("Enter City: ");
        fgets(s[i].city, sizeof(s[i].city), stdin);
        s[i].city[strcspn(s[i].city, "\n")] = '\0';

        printf("Enter Dream / Career Ambition: ");
        fgets(s[i].dream, sizeof(s[i].dream), stdin);
        s[i].dream[strcspn(s[i].dream, "\n")] = '\0';
    }

    printf("\n========== STUDENT RECORDS ==========\n");

    for (i = 0; i < n; i++)
    {
        printf("\nStudent %d\n", i + 1);
        printf("Roll Number      : %d\n", s[i].rollNo);
        printf("Name             : %s\n", s[i].name);
        printf("Cut-off Mark     : %.2f\n", s[i].cutoffMark);
        printf("City             : %s\n", s[i].city);
        printf("Dream / Ambition : %s\n", s[i].dream);
    }

    printf("\nTotal students: %d\n", n);
    printf("Records displayed successfully.\n");

    return 0;
}
